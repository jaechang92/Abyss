# 랜드마크 — 방을 구별시키는 큰 물건
#
# 🔑 왜 필요한가 — 스테이지당 9~10방을 도는데 방들이 같은 타일셋과 배경을 공유한다.
#    랜드마크가 없으면 플레이어는 **자기가 어디까지 왔는지 모른다.**
#    미니맵이 있어도 마찬가지다 — 미니맵은 위치를 알려주지 위치를 기억시키지는 않는다.
#    방 하나에 하나씩 큰 것이 서 있으면 그게 그 방의 이름이 된다.
#
# 💡 소설이 이걸 그대로 한다 — 하란은 자리를 세어 방을 구별한다.
#    닳은 돌 아홉, 빈 집 열아홉, 먼지 두께 넷(3권 18·20장). **자리가 이름을 대신한다.**
#
# 📌 로비에서는 용도가 다르다. 여기서는 랜드마크가 **NPC 4명의 자리 표식**이다
#    (기록자·각인사·안내자·제단지기). 그래서 로비 COUNT 는 4가 필요하고 넷이 확실히 달라야 한다.
#
# ⚠️ 크지만 조용해야 한다. 랜드마크는 기억되기 위한 것이지 상호작용을 유도하기 위한 것이
#    아니다 — 만질 수 있는 것은 prop_interactive 쪽이고, 둘이 헷갈리면 어포던스가 무너진다.
# ══════════════════════════════════════════════════════════════

[KIND] prop
[PARAMS] PARAMS-PROP
[USE-SCENE] MATERIAL
[USE-MOTIF] yes
[USE-TONE] no
[USE-VOICE] SILENT
[COUNT] 4

[OVERRIDE]
image_size = 96x96
detail = high

[BODY]
one large distinctive thing standing alone, tall enough to mark the room
