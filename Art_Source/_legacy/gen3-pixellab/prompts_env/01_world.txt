[CORE]
# 전 파트에 들어가는 최소 공통. 짧게 유지할 것 — 여기가 길면 파트 문구가 밀린다.
dark fantasy underground ruins

[MOTIF]
# 🔴 세계를 하나로 묶는 세 가지. 파트마다 이 중 하나만 골라 쓴다(전부 넣으면 희석된다).
#
#   ① fallen  — 이 세계의 모든 것은 위에서 내려왔다. 방향성이 항상 아래다.
#               (프롤로그 "떨어지는 동안에도 나는 아직 나였다")
#   ② carved  — 각인사가 파편에 이름을 새긴다. 벽에도 누군가 새겼다.
#               (10-narrative-plan §2-2 — 이름이 이 게임의 서사 화폐다)
#   ③ unclimbed — 아무도 못 올라갔다. 위를 향하다 만 것들만 남았다.
#               (기록자 = 그 자신도 올라가지 못한 자)
fallen = rubble fallen from above
carved = names scratched into the stone by hand
unclimbed = stairs that stop halfway up

[TONE]
# 00-concept §5 — 무거운 철학이 아니라 절제된 현실감이다.
# 여기 있던 것들은 대단한 존재가 아니라 사람이었다.
quiet, melancholy, not heroic

blue sky, daylight, grass, text, watermark, ui
